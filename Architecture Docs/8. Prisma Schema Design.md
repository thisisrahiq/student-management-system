# 8. Prisma Schema Design

## 8.1 Overview

This document contains the complete, production-ready `schema.prisma` configuration file for the **Student Management System — Registry Module**.

It leverages Prisma ORM features including explicit relations, enums, auto-generated CUID identifiers, cascade deletion rules, field attributes (`@unique`, `@default`, `@updatedAt`), and index optimization.

---

## 8.2 Production `schema.prisma` Code

```prisma
// Architecture Docs / 8. Prisma Schema Design
// Generator & Datasource Configuration

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ==========================================
// ENUMS
// ==========================================

enum EnrolmentStatus {
  ENROLLED
  DEFERRED
  WITHDRAWN
  COMPLETED
}

enum GradeClassification {
  DISTINCTION
  MERIT
  PASS
  FAIL
}

// ==========================================
// MODELS
// ==========================================

model Programme {
  id        String   @id @default(cuid())
  code      String   @unique
  name      String   @unique
  baseFee   Decimal  @db.Decimal(10, 2)
  duration  String   // e.g. "3 Years", "1 Year Full-Time"
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  students  Student[]
  modules   Module[]

  @@map("Programmes")
}

model Student {
  id           String          @id @default(cuid())
  studentId    String          @unique // e.g. "SMS-2025-0001"
  fullName     String
  email        String          @unique
  dateOfBirth  DateTime
  academicYear String          // e.g. "2025/2026"
  status       EnrolmentStatus @default(ENROLLED)
  programmeId  String
  createdAt    DateTime        @default(now())
  updatedAt    DateTime        @updatedAt

  programme    Programme       @relation(fields: [programmeId], references: [id], onDelete: Restrict)
  fee          Fee?
  payments     Payment[]
  submissions  Submission[]

  @@index([studentId])
  @@index([fullName])
  @@index([status])
  @@index([academicYear])
  @@map("Students")
}

model Fee {
  id             String   @id @default(cuid())
  studentId      String   @unique
  assignedAmount Decimal  @db.Decimal(10, 2)
  totalPaid      Decimal  @default(0.00) @db.Decimal(10, 2)
  createdAt      DateTime @default(now())
  updatedAt      DateTime @updatedAt

  student        Student   @relation(fields: [studentId], references: [id], onDelete: Cascade)
  payments       Payment[]

  @@map("Fees")
}

model Payment {
  id              String   @id @default(cuid())
  studentId       String
  feeId           String
  amount          Decimal  @db.Decimal(10, 2)
  paymentDate     DateTime @default(now())
  referenceNumber String   @unique // e.g. "PAY-99821"
  createdAt       DateTime @default(now())

  student         Student  @relation(fields: [studentId], references: [id], onDelete: Cascade)
  fee             Fee      @relation(fields: [feeId], references: [id], onDelete: Cascade)

  @@index([studentId])
  @@index([paymentDate])
  @@map("Payments")
}

model Module {
  id          String   @id @default(cuid())
  code        String   @unique // e.g. "CS101"
  title       String
  programmeId String
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  programme   Programme    @relation(fields: [programmeId], references: [id], onDelete: Restrict)
  assessments Assessment[]

  @@map("Modules")
}

model Assessment {
  id        String   @id @default(cuid())
  title     String
  moduleId  String
  deadline  DateTime
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  module      Module       @relation(fields: [moduleId], references: [id], onDelete: Cascade)
  submissions Submission[]

  @@index([deadline])
  @@map("Assessments")
}

model Submission {
  id           String   @id @default(cuid())
  assessmentId String
  studentId    String
  filePath     String   // Local path e.g. "/uploads/UUID_assignment1.pdf"
  fileName     String   // Original filename e.g. "John_Doe_CS101.pdf"
  submittedAt  DateTime @default(now())
  isLate       Boolean  @default(false)
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  assessment   Assessment @relation(fields: [assessmentId], references: [id], onDelete: Cascade)
  student      Student    @relation(fields: [studentId], references: [id], onDelete: Cascade)
  result       Result?

  @@unique([assessmentId, studentId]) // Enforces 1 active submission record per student per assessment
  @@index([studentId])
  @@index([assessmentId])
  @@map("Submissions")
}

model Result {
  id             String              @id @default(cuid())
  submissionId   String              @unique
  numericGrade   Int                 // 0 - 100
  classification GradeClassification
  isPublished    Boolean             @default(false)
  gradedAt       DateTime            @default(now())
  updatedAt      DateTime            @updatedAt

  submission     Submission          @relation(fields: [submissionId], references: [id], onDelete: Cascade)

  @@index([isPublished])
  @@map("Results")
}
```

---

## 8.3 Key Prisma Schema Features

1. **Strict Type Safety**: TypeScript types automatically generated for client applications match enums and database structures.
2. **Resubmission Constraint**: `@@unique([assessmentId, studentId])` ensures every student has at most one active submission per assessment, enabling seamless upsert behavior during resubmission.
3. **Withheld Confidentiality**: `isPublished` Boolean flag with index `@index([isPublished])` allows high-performance server-side filtering for student marksheets.
4. **Cascading Deletions**: Deleting a student cascades deletion to their fee record, payments, and submissions, maintaining total data cleanliness.
